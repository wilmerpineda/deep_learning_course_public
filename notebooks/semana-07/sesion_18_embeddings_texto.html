Sesión 18 — Embeddings para texto

Curso: Deep Learning · Universidad Externado de Colombia

Propósito y evidencia

Verificaremos que un embedding es un lookup entrenable, inspeccionaremos la máscara y aprenderemos una geometría pequeña y dependiente de tarea.

import os
os.environ["TF_CPP_MIN_LOG_LEVEL"] = "2"
from collections import Counter
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from sklearn.decomposition import PCA

SEED = 42
keras.utils.set_random_seed(SEED)
np.set_printoptions(precision=5, suppress=True)
print("TensorFlow:", tf.__version__)

1. Vocabulario, índices y padding

El índice 0 se reserva para padding y 1 para desconocidos. Los demás índices identifican tokens; su valor numérico no expresa similitud.

sentences = [
    "precio del café sube", "exportaciones crecen este mes",
    "ventas aumentan con fuerza", "demanda mejora hoy",
    "volumen exportado sube", "mercado muestra crecimiento",
    "precio alcanza nivel alto", "envíos crecen en junio",
    "producción aumenta este mes", "ingresos mejoran hoy",
    "precio del café cae", "exportaciones bajan este mes",
    "ventas disminuyen con fuerza", "demanda empeora hoy",
    "volumen exportado cae", "mercado muestra descenso",
    "precio alcanza nivel bajo", "envíos bajan en junio",
    "producción disminuye este mes", "ingresos empeoran hoy",
]
labels = np.array([1] * 10 + [0] * 10, dtype=np.float32)
tokens = [word for sentence in sentences for word in sentence.split()]
vocabulary = ["<pad>", "<unk>"] + sorted(set(tokens))
token_to_id = {token: i for i, token in enumerate(vocabulary)}
id_to_token = dict(enumerate(vocabulary))
MAX_LEN = 6

def encode(sentence):
    ids = [token_to_id.get(token, 1) for token in sentence.split()][:MAX_LEN]
    return ids + [0] * (MAX_LEN - len(ids))

X = np.array([encode(sentence) for sentence in sentences], dtype=np.int32)
print("vocabulario:", len(vocabulary), "shape:", X.shape)
display(pd.DataFrame(X[:4], columns=[f"t{i}" for i in range(MAX_LEN)]))

2. Lookup y one-hot son equivalentes

vocab_size, embedding_dim = len(vocabulary), 4
rng = np.random.default_rng(SEED)
E = rng.normal(size=(vocab_size, embedding_dim)).astype(np.float32)
selected_ids = np.array([token_to_id["café"], token_to_id["precio"]])
lookup = E[selected_ids]
one_hot = np.eye(vocab_size, dtype=np.float32)[selected_ids]
product = one_hot @ E
np.testing.assert_allclose(lookup, product)
print("error máximo:", np.max(np.abs(lookup - product)))
display(pd.DataFrame(lookup, index=["café", "precio"]))

3. Shapes, parámetros y máscara

La capa convierte (batch, time) en (batch, time, embedding_dim). La máscara separa pasos válidos del padding.

embedding = keras.layers.Embedding(
    input_dim=vocab_size, output_dim=embedding_dim, mask_zero=True)
vectors = embedding(X[:2])
mask = embedding.compute_mask(X[:2])
print("índices:", X[:2].shape, "vectores:", vectors.shape, "máscara:", mask.shape)
display(pd.DataFrame(mask.numpy(), columns=[f"t{i}" for i in range(MAX_LEN)]))
assert embedding.count_params() == vocab_size * embedding_dim

4. El padding no participa en el promedio

pool = keras.layers.GlobalAveragePooling1D()
pooled = pool(vectors, mask=mask).numpy()
manual = np.array([
    row_vectors[row_mask].mean(axis=0)
    for row_vectors, row_mask in zip(vectors.numpy(), mask.numpy())
])
np.testing.assert_allclose(pooled, manual, rtol=1e-6, atol=1e-6)
print("error promedio enmascarado:", np.max(np.abs(pooled - manual)))

5. Una tarea pequeña organiza la tabla

La etiqueta indica aumento o disminución. El corpus permite observar el mecanismo, pero no pretende ser un modelo de lenguaje.

train_idx = np.array([0,1,2,3,4,5,6,8,10,11,12,13,14,15,16,18])
test_idx = np.array([7,9,17,19])
keras.utils.set_random_seed(SEED)
model = keras.Sequential([
    keras.Input(shape=(MAX_LEN,), dtype="int32"),
    keras.layers.Embedding(vocab_size, 6, mask_zero=True, name="embedding"),
    keras.layers.GlobalAveragePooling1D(),
    keras.layers.Dense(1, activation="sigmoid"),
])
model.compile(optimizer=keras.optimizers.Adam(0.03),
              loss="binary_crossentropy", metrics=["accuracy"])
model.fit(
    X[train_idx], labels[train_idx], validation_split=0.25,
    epochs=80, batch_size=4, verbose=0,
    callbacks=[keras.callbacks.EarlyStopping(
        monitor="val_loss", patience=10, restore_best_weights=True
    )],
)
_, accuracy = model.evaluate(X[test_idx], labels[test_idx], verbose=0)
print("accuracy test:", round(float(accuracy), 3))
print("parámetros totales:", model.count_params())

6. Vecinos por similitud coseno

La consulta excluye padding, desconocido y la propia palabra. Los resultados son exploratorios porque el corpus es pequeño.

learned = model.get_layer("embedding").get_weights()[0]

def nearest(token, k=3):
    idx = token_to_id[token]
    normalized = learned / np.maximum(np.linalg.norm(learned, axis=1, keepdims=True), 1e-12)
    scores = normalized @ normalized[idx]
    order = [i for i in np.argsort(scores)[::-1] if i not in {0, 1, idx}][:k]
    return pd.DataFrame({
        "token": [id_to_token[i] for i in order],
        "coseno": [scores[i] for i in order],
    })

print("Vecinos de sube")
display(nearest("sube"))
print("Vecinos de cae")
display(nearest("cae"))

7. PCA proyecta; no demuestra significado

tokens_to_plot = [
    "sube", "crecen", "aumentan", "mejora",
    "cae", "bajan", "disminuyen", "empeora",
    "precio", "exportaciones",
]
indices = [token_to_id[token] for token in tokens_to_plot]
coords = PCA(n_components=2).fit_transform(learned[indices])
fig, ax = plt.subplots(figsize=(8, 5))
ax.scatter(coords[:, 0], coords[:, 1], color="#0f766e")
for token, point in zip(tokens_to_plot, coords):
    ax.annotate(token, point, xytext=(4, 4), textcoords="offset points")
ax.axhline(0, color="gray", linewidth=0.7)
ax.axvline(0, color="gray", linewidth=0.7)
ax.set_title("Proyección PCA del embedding aprendido")
ax.set_xlabel("PC1")
ax.set_ylabel("PC2")
plt.show()
