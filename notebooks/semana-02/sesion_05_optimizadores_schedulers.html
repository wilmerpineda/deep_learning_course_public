Sesión 5 — Optimizadores y schedulers

Resultado. Comparar la memoria y trayectoria de SGD, momentum, RMSProp y Adam, y justificar cuándo modificar la tasa de aprendizaje.

1. Preparación reproducible

Se controlan datos, inicialización, orden de batches, pérdida y presupuesto. El objetivo es explicar diferencias, no declarar un ganador universal.

3. Actualizaciones manuales en dos coordenadas

Usamos la misma secuencia de gradientes para hacer visible cómo cada método transforma el historial.

gradients = np.array([[4., .4], [3., -.3], [2., .25], [1., -.2]])
alpha, beta, rho, eps = .1, .9, .9, 1e-8
theta_sgd = np.zeros(2); theta_mom = np.zeros(2); theta_rms = np.zeros(2)
velocity = np.zeros(2); square_avg = np.zeros(2)
rows = []
for t, g in enumerate(gradients, start=1):
    theta_sgd -= alpha * g
    velocity = beta * velocity + g
    theta_mom -= alpha * velocity
    square_avg = rho * square_avg + (1 - rho) * g**2
    theta_rms -= alpha * g / (np.sqrt(square_avg) + eps)
    rows.append([t, g.copy(), theta_sgd.copy(), velocity.copy(), theta_mom.copy(), square_avg.copy(), theta_rms.copy()])
pd.DataFrame(rows, columns=['t', 'g_t', 'theta_SGD', 'v_t', 'theta_momentum', 'r_t', 'theta_RMSProp'])

4. Corrección de sesgo en Adam

Con un gradiente constante, los promedios sin corregir empiezan cerca de cero. La corrección recupera los momentos esperados desde el primer paso.

g = np.array([2.0, -3.0])
beta1, beta2 = .9, .999
m = np.zeros_like(g); v = np.zeros_like(g)
adam_rows = []
for t in range(1, 6):
    m = beta1 * m + (1 - beta1) * g
    v = beta2 * v + (1 - beta2) * g**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    adam_rows.append([t, m.copy(), m_hat.copy(), v.copy(), v_hat.copy()])
pd.DataFrame(adam_rows, columns=['t', 'm_t', 'm_hat', 'v_t', 'v_hat'])

5. Protocolo controlado con el mismo MLP

Test permanece fuera del experimento. Cada optimizador recibe los mismos pesos y la misma secuencia de batches.

X, y = make_moons(n_samples=600, noise=.24, random_state=RANDOM_STATE)
X_train, X_val, y_train, y_val = train_test_split(X, y, test_size=.25, stratify=y, random_state=RANDOM_STATE)
scaler = StandardScaler().fit(X_train)
X_train = scaler.transform(X_train).astype('float32')
X_val = scaler.transform(X_val).astype('float32')
y_train = y_train.astype('float32'); y_val = y_val.astype('float32')
order = np.random.default_rng(RANDOM_STATE).permutation(len(X_train))
X_train, y_train = X_train[order], y_train[order]

def build_model():
    return keras.Sequential([keras.Input(shape=(2,)), keras.layers.Dense(8, activation='tanh'), keras.layers.Dense(1, activation='sigmoid')])

keras.utils.set_random_seed(RANDOM_STATE)
template = build_model()
initial_weights = template.get_weights()
print({'train': X_train.shape, 'validación': X_val.shape, 'parámetros': template.count_params()})

6. Registro explícito de la tasa

El callback observa la tasa efectiva al final de cada época para que la política forme parte de la evidencia.

class LearningRateLogger(keras.callbacks.Callback):
    def on_train_begin(self, logs=None):
        self.values = []
    def on_epoch_end(self, epoch, logs=None):
        lr = self.model.optimizer.learning_rate
        if callable(lr):
            lr = lr(self.model.optimizer.iterations)
        self.values.append(float(tf.keras.backend.get_value(lr)))

def train_optimizer(name, optimizer, epochs=35, callbacks=None):
    model = build_model()
    model.set_weights([w.copy() for w in initial_weights])
    logger = LearningRateLogger()
    callback_list = [logger] + (callbacks or [])
    model.compile(optimizer=optimizer, loss='binary_crossentropy', metrics=['accuracy'])
    history = model.fit(X_train, y_train, validation_data=(X_val, y_val), epochs=epochs, batch_size=32, shuffle=False, verbose=0, callbacks=callback_list)
    frame = pd.DataFrame(history.history)
    frame['epoca'] = np.arange(1, len(frame) + 1)
    frame['learning_rate'] = logger.values
    frame['optimizador'] = name
    return frame

runs = {
    'SGD': train_optimizer('SGD', keras.optimizers.SGD(learning_rate=.05)),
    'Momentum': train_optimizer('Momentum', keras.optimizers.SGD(learning_rate=.05, momentum=.9)),
    'RMSProp': train_optimizer('RMSProp', keras.optimizers.RMSprop(learning_rate=.001)),
    'Adam': train_optimizer('Adam', keras.optimizers.Adam(learning_rate=.001)),
}
fig, axes = plt.subplots(1, 2, figsize=(11, 4))
for name, frame in runs.items():
    axes[0].plot(frame.epoca, frame.loss, label=name)
    axes[1].plot(frame.epoca, frame.val_loss, label=name)
axes[0].set_title('Entrenamiento'); axes[1].set_title('Validación')
for ax in axes: ax.set_xlabel('época'); ax.set_ylabel('BCE'); ax.grid(alpha=.25); ax.legend()
plt.tight_layout(); plt.show()

pd.DataFrame([{
    'optimizador': name,
    'val_loss_final': float(frame.val_loss.iloc[-1]),
    'mejor_val_loss': float(frame.val_loss.min()),
    'época_mejor': int(frame.val_loss.idxmin() + 1),
} for name, frame in runs.items()])

7. Schedule predeterminado y callback reactivo

Comparamos exponential decay, definido por el paso, con ReduceLROnPlateau, que reacciona a validación.

steps_per_epoch = int(np.ceil(len(X_train) / 32))
decay = keras.optimizers.schedules.ExponentialDecay(initial_learning_rate=.05, decay_steps=8 * steps_per_epoch, decay_rate=.5, staircase=True)
scheduled = train_optimizer('SGD + decay', keras.optimizers.SGD(learning_rate=decay))
plateau_callback = keras.callbacks.ReduceLROnPlateau(monitor='val_loss', factor=.5, patience=3, min_delta=1e-3, min_lr=1e-4)
plateau = train_optimizer('SGD + plateau', keras.optimizers.SGD(learning_rate=.05), callbacks=[plateau_callback])

fig, axes = plt.subplots(1, 2, figsize=(11, 4))
for frame in [scheduled, plateau]:
    label = frame.optimizador.iloc[0]
    axes[0].plot(frame.epoca, frame.learning_rate, label=label)
    axes[1].plot(frame.epoca, frame.val_loss, label=label)
axes[0].set_title('Tasa observada'); axes[1].set_title('Pérdida de validación')
for ax in axes: ax.set_xlabel('época'); ax.grid(alpha=.25); ax.legend()
plt.tight_layout(); plt.show()

8. Cierre y recomendación

Para cada configuración, documente: rapidez inicial, mejor validación, estabilidad, estado adicional y sensibilidad aparente a la tasa.
