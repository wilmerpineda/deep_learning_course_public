Propósito y evidencia

Calcularemos un paso LSTM manual, lo contrastaremos con LSTMCell y compararemos persistencia estacional, SimpleRNN y LSTM bajo un protocolo común. Si el archivo privado no está disponible, la parte manual y el caso sintético siguen funcionando.

1. Un paso LSTM manual

Keras concatena las puertas en el orden input, forget, candidate, output.

def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-z))

x_t = np.array([[0.4]], dtype=np.float32)
h_prev = np.array([[0.1, -0.2]], dtype=np.float32)
c_prev = np.array([[0.3, -0.1]], dtype=np.float32)
units = 2
rng = np.random.default_rng(7)
kernel = rng.normal(0, 0.25, size=(1, 4 * units)).astype(np.float32)
recurrent_kernel = rng.normal(0, 0.25, size=(units, 4 * units)).astype(np.float32)
bias = rng.normal(0, 0.05, size=(4 * units,)).astype(np.float32)

z = x_t @ kernel + h_prev @ recurrent_kernel + bias
z_i, z_f, z_c, z_o = np.split(z, 4, axis=1)
i, f, candidate, o = sigmoid(z_i), sigmoid(z_f), np.tanh(z_c), sigmoid(z_o)
c_manual = f * c_prev + i * candidate
h_manual = o * np.tanh(c_manual)
display(pd.DataFrame({
    "puerta": ["input", "forget", "candidate", "output"],
    "unidad_1": [i[0, 0], f[0, 0], candidate[0, 0], o[0, 0]],
    "unidad_2": [i[0, 1], f[0, 1], candidate[0, 1], o[0, 1]],
}).round(6))
print("h_t:", h_manual, "c_t:", c_manual)

2. Verificación con LSTMCell

cell = keras.layers.LSTMCell(units)
_ = cell(x_t, states=[h_prev, c_prev])
cell.set_weights([kernel, recurrent_kernel, bias])
h_keras, [h_state, c_state] = cell(x_t, states=[h_prev, c_prev], training=False)
np.testing.assert_allclose(h_manual, h_keras.numpy(), rtol=1e-6, atol=1e-6)
np.testing.assert_allclose(c_manual, c_state.numpy(), rtol=1e-6, atol=1e-6)
expected_params = 4 * (1 * units + units * units + units)
assert cell.count_params() == expected_params == 32
print("error h:", np.max(np.abs(h_manual - h_keras.numpy())))
print("parámetros:", cell.count_params())

3. Dependencia sintética larga

La etiqueta depende de los tres primeros pasos y los restantes añaden ruido. Es una condición controlada, no una prueba de superioridad universal.

rng = np.random.default_rng(SEED)
n, timesteps = 600, 30
X = rng.normal(size=(n, timesteps, 1)).astype(np.float32)
y = (X[:, :3, 0].sum(axis=1) > 0).astype(np.float32)
X_train, y_train = X[:420], y[:420]
X_val, y_val = X[420:510], y[420:510]
X_test_syn, y_test_syn = X[510:], y[510:]

def sequence_classifier(kind, seed):
    keras.utils.set_random_seed(seed)
    recurrent = keras.layers.SimpleRNN(8) if kind == "SimpleRNN" else keras.layers.LSTM(8)
    model = keras.Sequential([
        keras.Input(shape=(timesteps, 1)), recurrent,
        keras.layers.Dense(1, activation="sigmoid"),
    ])
    model.compile(optimizer=keras.optimizers.Adam(0.01),
                  loss="binary_crossentropy", metrics=["accuracy"])
    return model

synthetic_rows = []
for kind in ("SimpleRNN", "LSTM"):
    model = sequence_classifier(kind, SEED)
    model.fit(
        X_train, y_train, validation_data=(X_val, y_val),
        epochs=25, batch_size=32, verbose=0,
        callbacks=[keras.callbacks.EarlyStopping(
            monitor="val_loss", patience=4, restore_best_weights=True
        )],
    )
    _, accuracy = model.evaluate(X_test_syn, y_test_syn, verbose=0)
    synthetic_rows.append({"modelo": kind, "accuracy_test": accuracy})
display(pd.DataFrame(synthetic_rows).round(3))

5. Ventanas y cortes sin fuga

El scaler aprende solo de train. Las ventanas se construyen sobre toda la cronología ya transformada y se asignan según la fecha del target.

LOOK_BACK = 12
TRAIN_END = pd.Timestamp("2013-09-01")
VAL_END = pd.Timestamp("2014-09-01")

def make_windows(values, dates, look_back=12):
    X, y, target_dates, positions = [], [], [], []
    for target_pos in range(look_back, len(values)):
        X.append(values[target_pos - look_back:target_pos])
        y.append(values[target_pos])
        target_dates.append(dates.iloc[target_pos])
        positions.append(target_pos)
    return (np.asarray(X, dtype=np.float32), np.asarray(y, dtype=np.float32),
            pd.DatetimeIndex(target_dates), np.asarray(positions))

if coffee is not None:
    raw = coffee[["Exportaciones"]].to_numpy(dtype=np.float32)
    scaler = MinMaxScaler().fit(raw[coffee["Fechas"] <= TRAIN_END])
    scaled = scaler.transform(raw)
    X_all, y_all, target_dates, positions = make_windows(
        scaled, coffee["Fechas"], LOOK_BACK
    )
    train_mask = target_dates <= TRAIN_END
    val_mask = (target_dates > TRAIN_END) & (target_dates <= VAL_END)
    test_mask = target_dates > VAL_END
    splits = {
        "train": (X_all[train_mask], y_all[train_mask]),
        "validación": (X_all[val_mask], y_all[val_mask]),
        "test": (X_all[test_mask], y_all[test_mask]),
    }
    print({name: (X.shape, y.shape) for name, (X, y) in splits.items()})
    assert splits["validación"][0].shape[0] == 12
    assert splits["test"][0].shape[0] == 3
    assert target_dates[test_mask].strftime("%Y-%m").tolist() == [
        "2014-10", "2014-11", "2014-12"
    ]

6. Baseline y redes con el mismo presupuesto

La persistencia estacional usa el valor de doce meses atrás. Ambas redes comparten cortes, ventana, unidades, optimizador, batch, épocas y semillas.

def rmse(y_true, y_pred):
    return float(np.sqrt(mean_squared_error(y_true, y_pred)))

def build_forecaster(kind, seed, look_back=12):
    keras.utils.set_random_seed(seed)
    recurrent = keras.layers.SimpleRNN(4) if kind == "SimpleRNN" else keras.layers.LSTM(4)
    model = keras.Sequential([
        keras.Input(shape=(look_back, 1)), recurrent, keras.layers.Dense(1),
    ])
    model.compile(optimizer=keras.optimizers.Adam(0.01), loss="mse")
    return model

coffee_results = coffee_predictions = None
if coffee is not None:
    X_train_c, y_train_c = splits["train"]
    X_val_c, y_val_c = splits["validación"]
    X_test_c, y_test_c = splits["test"]
    y_test_real = scaler.inverse_transform(y_test_c).ravel()
    test_positions = positions[test_mask]
    seasonal_pred = raw[test_positions - LOOK_BACK, 0]
    rows = [{"modelo": "Persistencia estacional", "semilla": "—",
             "MAE_test": mean_absolute_error(y_test_real, seasonal_pred),
             "RMSE_test": rmse(y_test_real, seasonal_pred)}]
    prediction_map = {"Real": y_test_real, "Persistencia estacional": seasonal_pred}

    for kind in ("SimpleRNN", "LSTM"):
        per_seed = []
        for seed in (17, 42, 73):
            model = build_forecaster(kind, seed)
            model.fit(
                X_train_c, y_train_c, validation_data=(X_val_c, y_val_c),
                epochs=80, batch_size=16, verbose=0,
                callbacks=[keras.callbacks.EarlyStopping(
                    monitor="val_loss", patience=8, restore_best_weights=True
                )],
            )
            pred = scaler.inverse_transform(model.predict(X_test_c, verbose=0)).ravel()
            per_seed.append(pred)
            rows.append({"modelo": kind, "semilla": seed,
                         "MAE_test": mean_absolute_error(y_test_real, pred),
                         "RMSE_test": rmse(y_test_real, pred)})
        prediction_map[kind] = np.mean(per_seed, axis=0)

    coffee_results = pd.DataFrame(rows)
    coffee_predictions = pd.DataFrame(prediction_map, index=target_dates[test_mask])
    display(coffee_results.round(2))
    display(coffee_predictions.round(2))
