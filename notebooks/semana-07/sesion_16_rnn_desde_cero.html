Sesión 16 — RNN desde cero

Curso: Deep Learning · Universidad Externado de Colombia

Propósito y evidencia

Calcularemos tres estados con NumPy y asignaremos exactamente los mismos pesos a keras.layers.SimpleRNN. La evidencia será la traza temporal, los shapes, el conteo de parámetros y el error máximo NumPy–Keras.

import os
os.environ["TF_CPP_MIN_LOG_LEVEL"] = "2"
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras

SEED = 42
keras.utils.set_random_seed(SEED)
np.set_printoptions(precision=6, suppress=True)
print("TensorFlow:", tf.__version__)

1. Entrada secuencial

Usaremos un batch, tres pasos y una feature. Keras ordena los ejes como (batch, time, features).

x = np.array([[[0.20], [0.50], [-0.10]]], dtype=np.float32)
batch, time, features = x.shape
print("shape:", x.shape)
assert x.shape == (1, 3, 1)

2. Pesos compartidos y libro contable

Con una feature y dos unidades, el conteo es du + u² + u = 8. La longitud temporal cambia el número de aplicaciones, no los parámetros.

W_x = np.array([[0.70, -0.40]], dtype=np.float32)
W_h = np.array([[0.50, 0.10], [-0.30, 0.80]], dtype=np.float32)
b = np.array([0.05, -0.02], dtype=np.float32)
units = W_h.shape[0]
manual_params = features * units + units * units + units
print("parámetros:", manual_params)
assert manual_params == 8

3. Forward manual

En cada paso guardamos la contribución de la entrada, la recurrencia, la preactivación y el estado.

def manual_simple_rnn(sequence, kernel, recurrent_kernel, bias):
    h = np.zeros((sequence.shape[0], recurrent_kernel.shape[0]), dtype=np.float32)
    states, rows = [], []
    for t in range(sequence.shape[1]):
        input_term = sequence[:, t, :] @ kernel
        recurrent_term = h @ recurrent_kernel
        preactivation = input_term + recurrent_term + bias
        h = np.tanh(preactivation)
        states.append(h.copy())
        rows.append({
            "t": t + 1, "x_t": float(sequence[0, t, 0]),
            "entrada_1": float(input_term[0, 0]),
            "recurrencia_1": float(recurrent_term[0, 0]),
            "a_1": float(preactivation[0, 0]),
            "h_1": float(h[0, 0]), "h_2": float(h[0, 1]),
        })
    return np.stack(states, axis=1), pd.DataFrame(rows)

manual_states, trace = manual_simple_rnn(x, W_x, W_h, b)
display(trace.round(6))
print("estados:", manual_states.shape)

4. La misma celda en Keras

return_sequences conserva todos los estados y return_state añade el estado final.

layer = keras.layers.SimpleRNN(
    units=units, activation="tanh",
    return_sequences=True, return_state=True,
)
_ = layer(x)
layer.set_weights([W_x, W_h, b])
keras_states, keras_final = layer(x, training=False)
keras_states, keras_final = keras_states.numpy(), keras_final.numpy()

np.testing.assert_allclose(manual_states, keras_states, rtol=1e-6, atol=1e-6)
np.testing.assert_allclose(manual_states[:, -1], keras_final, rtol=1e-6, atol=1e-6)
assert layer.count_params() == manual_params
print("error máximo:", np.max(np.abs(manual_states - keras_states)))
print("count_params:", layer.count_params())

5. Muchos a uno y muchos a muchos

last_only = keras.layers.SimpleRNN(units=2, return_sequences=False)
whole_sequence = keras.layers.SimpleRNN(units=2, return_sequences=True)
print("muchos a uno:", last_only(x).shape)
print("muchos a muchos:", whole_sequence(x).shape)
assert last_only(x).shape == (1, 2)
assert whole_sequence(x).shape == (1, 3, 2)

6. Productos repetidos y distancia temporal

En la recurrencia escalar lineal h_t = r h_(t-1) + x_t, la sensibilidad de h_T respecto a h_0 es r^T. Esta simplificación hace visible el producto repetido.

steps = np.arange(1, 41)
for r in (0.7, 0.95, 1.05):
    plt.plot(steps, np.abs(r) ** steps, label=f"|r|={r}")
plt.axhline(1.0, color="black", linewidth=1)
plt.yscale("log")
plt.xlabel("distancia temporal")
plt.ylabel("magnitud de la sensibilidad")
plt.title("Productos repetidos: desvanecimiento y explosión")
plt.legend()
plt.show()
