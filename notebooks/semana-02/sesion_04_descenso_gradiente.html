2. Una trayectoria que podemos auditar

Para \(J(w_1,w_2)=(w_1-2)^2+4(w_2+1)^2\), el gradiente es \([2(w_1-2),8(w_2+1)]^T\).

def objective(w):
    return (w[0] - 2.0)**2 + 4.0 * (w[1] + 1.0)**2

def gradient(w):
    return np.array([2.0 * (w[0] - 2.0), 8.0 * (w[1] + 1.0)])

def descend(alpha, steps=18, start=(-3.0, 2.0)):
    w = np.array(start, dtype=float)
    rows = []
    for t in range(steps + 1):
        rows.append((t, *w, objective(w), np.linalg.norm(gradient(w))))
        w = w - alpha * gradient(w)
    return pd.DataFrame(rows, columns=['paso', 'w1', 'w2', 'J', 'norma_gradiente'])

toy_runs = {alpha: descend(alpha) for alpha in [0.03, 0.12, 0.26]}
pd.concat({str(k): v.tail(1) for k, v in toy_runs.items()}, names=['alpha'])
fig, axes = plt.subplots(1, 2, figsize=(11, 4))
for alpha, run in toy_runs.items():
    axes[0].plot(run.paso, run.J, marker='o', ms=3, label=f'α={alpha}')
    axes[1].plot(run.w1, run.w2, marker='o', ms=3, label=f'α={alpha}')
axes[0].set_yscale('symlog'); axes[0].set_title('Objetivo por actualización')
axes[1].scatter([2], [-1], marker='*', s=160, c='black', label='mínimo')
axes[1].set_title('Trayectoria en parámetros')
for ax in axes: ax.grid(alpha=.25); ax.legend()
plt.tight_layout(); plt.show()

3. Datos y particiones fijas

Test queda reservado. Las comparaciones de esta sesión usan únicamente entrenamiento y validación.

X, y = make_moons(n_samples=600, noise=.24, random_state=RANDOM_STATE)
X_train, X_val, y_train, y_val = train_test_split(X, y, test_size=.25, stratify=y, random_state=RANDOM_STATE)
scaler = StandardScaler().fit(X_train)
X_train = scaler.transform(X_train).astype('float32')
X_val = scaler.transform(X_val).astype('float32')
y_train = y_train.astype('float32')
y_val = y_val.astype('float32')
order = np.random.default_rng(RANDOM_STATE).permutation(len(X_train))
X_train, y_train = X_train[order], y_train[order]
print({'train': X_train.shape, 'validación': X_val.shape})

4. Misma inicialización, una decisión distinta

Creamos una plantilla de pesos. Cada ejecución empieza exactamente desde esos valores y usa el mismo orden de casos.

def build_model():
    return keras.Sequential([
        keras.Input(shape=(2,)),
        keras.layers.Dense(8, activation='tanh'),
        keras.layers.Dense(1, activation='sigmoid'),
    ])

keras.utils.set_random_seed(RANDOM_STATE)
template = build_model()
initial_weights = template.get_weights()

def train_run(learning_rate, batch_size, epochs=35):
    model = build_model()
    model.set_weights([w.copy() for w in initial_weights])
    model.compile(optimizer=keras.optimizers.SGD(learning_rate=learning_rate), loss='binary_crossentropy', metrics=['accuracy'])
    history = model.fit(X_train, y_train, validation_data=(X_val, y_val), epochs=epochs, batch_size=batch_size, shuffle=False, verbose=0)
    frame = pd.DataFrame(history.history)
    frame['epoca'] = np.arange(1, len(frame) + 1)
    frame['alpha'] = learning_rate
    frame['batch'] = batch_size
    frame['pasos_por_epoca'] = int(np.ceil(len(X_train) / batch_size))
    return frame

runs_lr = {alpha: train_run(alpha, batch_size=32) for alpha in [.001, .05, 1.0]}
fig, axes = plt.subplots(1, 2, figsize=(11, 4))
for alpha, frame in runs_lr.items():
    axes[0].plot(frame.epoca, frame.loss, label=f'α={alpha}')
    axes[1].plot(frame.epoca, frame.val_loss, label=f'α={alpha}')
axes[0].set_title('Entrenamiento'); axes[1].set_title('Validación')
for ax in axes: ax.set_xlabel('época'); ax.set_ylabel('BCE'); ax.grid(alpha=.25); ax.legend()
plt.tight_layout(); plt.show()

5. El batch cambia el número de actualizaciones

Conservamos tasa y épocas para observar tanto la trayectoria como el presupuesto resultante.

runs_batch = {batch: train_run(.05, batch_size=batch) for batch in [8, 32, len(X_train)]}
summary = []
for batch, frame in runs_batch.items():
    summary.append({
        'batch': batch,
        'pasos_por_epoca': int(frame.pasos_por_epoca.iloc[0]),
        'actualizaciones_totales': int(frame.pasos_por_epoca.iloc[0] * len(frame)),
        'val_loss_final': float(frame.val_loss.iloc[-1]),
    })
pd.DataFrame(summary)
plt.figure(figsize=(7, 4))
for batch, frame in runs_batch.items():
    steps = frame.epoca * frame.pasos_por_epoca
    plt.plot(steps, frame.val_loss, label=f'B={batch}')
plt.xlabel('actualizaciones acumuladas'); plt.ylabel('BCE de validación')
plt.grid(alpha=.25); plt.legend(); plt.show()
