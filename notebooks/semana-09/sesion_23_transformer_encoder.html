Sesión 23 — Encoder Transformer

Curso: Deep Learning · Universidad Externado de Colombia

Propósito y evidencia

Construiremos posición, varias cabezas, residuales y feed-forward hasta obtener un encoder Transformer auditable.

import os
os.environ["TF_CPP_MIN_LOG_LEVEL"]="2"
import numpy as np
import tensorflow as tf
from tensorflow import keras

SEED=42
keras.utils.set_random_seed(SEED)

1. Sin posición, permutar tokens permuta salidas

X=tf.constant([[[1.,0.,0.,1.],[0.,1.,1.,0.],[1.,1.,0.,0.]]])
mha=keras.layers.MultiHeadAttention(num_heads=2,key_dim=2)
Y=mha(X,X)
perm=[2,0,1]
Y_perm=mha(tf.gather(X,perm,axis=1),tf.gather(X,perm,axis=1))
np.testing.assert_allclose(tf.gather(Y,perm,axis=1),Y_perm,rtol=1e-5,atol=1e-5)
print("equivarianza verificada")

2. La posición suma una señal distinta a cada paso

def sinusoidal_position(length,dim):
    positions=np.arange(length)[:,None]
    rates=1/np.power(10000,(2*(np.arange(dim)//2))/dim)
    angles=positions*rates
    encoding=np.zeros((length,dim),dtype="float32")
    encoding[:,0::2]=np.sin(angles[:,0::2])
    encoding[:,1::2]=np.cos(angles[:,1::2])
    return encoding

P=sinusoidal_position(3,4)
print(P)
assert P.shape==(3,4)

3. Con posición, la permutación altera el problema

with_position=X+P[None,:,:]
permuted_tokens=tf.gather(X,perm,axis=1)+P[None,:,:]
A=mha(with_position,with_position)
B=mha(permuted_tokens,permuted_tokens)
print("salidas equivalentes:",np.allclose(tf.gather(A,perm,axis=1),B))
assert not np.allclose(tf.gather(A,perm,axis=1),B)

4. Cada cabeza dispone de su propia proyección

output,scores=mha(with_position,with_position,return_attention_scores=True)
print("entrada:",with_position.shape,"scores:",scores.shape,"salida:",output.shape)
assert scores.shape==(1,2,3,3)

5. Un encoder conserva el shape para habilitar residuales

def encoder_block(x,mask=None,d_model=4,heads=2,ff_dim=12):
    attention=keras.layers.MultiHeadAttention(num_heads=heads,key_dim=d_model//heads)(
        x,x,attention_mask=mask)
    x=keras.layers.LayerNormalization()(x+attention)
    hidden=keras.layers.Dense(ff_dim,activation="relu")(x)
    projected=keras.layers.Dense(d_model)(hidden)
    return keras.layers.LayerNormalization()(x+projected)

inputs=keras.Input((None,4))
outputs=encoder_block(inputs)
encoder=keras.Model(inputs,outputs)
encoded=encoder(with_position)
print("entrada/salida:",with_position.shape,encoded.shape,"parámetros:",encoder.count_params())
assert encoded.shape==with_position.shape

6. El libro contable detecta errores antes de entrenar

ledger={"X":tuple(with_position.shape),"QK scores":tuple(scores.shape),

7. La máscara debe llegar a cada bloque

valid=tf.constant([[True,True,False]])
attention_mask=tf.broadcast_to(valid[:,None,:],(1,3,3))
masked_mha=keras.layers.MultiHeadAttention(num_heads=2,key_dim=2)
_,masked_scores=masked_mha(with_position,with_position,
    attention_mask=attention_mask,return_attention_scores=True)
np.testing.assert_allclose(masked_scores.numpy()[...,2],0,atol=1e-7)
print("padding ignorado por ambas cabezas")

8. El vector final depende de token, posición y contexto

changed=tf.tensor_scatter_nd_update(with_position,[[0,1]],[tf.constant([2.,0.,1.,0.])])
encoded_changed=encoder(changed)
assert not np.allclose(encoded.numpy()[0,0],encoded_changed.numpy()[0,0])
print("el token 1 conserva su entrada, pero cambia su representación final")

Actividad y cierre

Dibuje el bloque, escriba los shapes y localice dónde una suma residual fallaría. Un Transformer no reemplaza el embedding: comienza con él y lo transforma en una representación contextual mediante posición, atención y capas por token.
