Taller 2 — Auditoría comparativa de embeddings

Construiremos una comparación controlada sobre texto real. El objetivo no es declarar un campeón, sino separar representación, clasificador y protocolo.

import os
os.environ["TF_CPP_MIN_LOG_LEVEL"] = "2"
from pathlib import Path
import re, time
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow import keras
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import f1_score, confusion_matrix

SEED = 42
keras.utils.set_random_seed(SEED)

1. Protocolo antes del modelo

paths = [Path("datasets/public/spanish_fake_news.csv"), Path("../../datasets/public/spanish_fake_news.csv")]
data = pd.read_csv(next(p for p in paths if p.exists())).fillna("")
data["document"] = (data.headline + " " + data.text).str.strip()
data["target"] = (data.label == "true").astype(int)
parts = {name: data[data.split == name].copy() for name in ["train", "validation", "test"]}
print({name: len(frame) for name, frame in parts.items()})

2. TF–IDF fija una referencia fuerte y barata

tfidf = TfidfVectorizer(max_features=2500, ngram_range=(1, 2), min_df=2)
X_train = tfidf.fit_transform(parts["train"].document)
X_test = tfidf.transform(parts["test"].document)
baseline = LogisticRegression(max_iter=500, random_state=SEED)
t0 = time.perf_counter(); baseline.fit(X_train, parts["train"].target); elapsed = time.perf_counter()-t0
pred = baseline.predict(X_test)
results = [{"modelo": "TF-IDF + LR", "macro_f1": f1_score(parts["test"].target, pred, average="macro"),
            "segundos": elapsed, "parámetros": X_train.shape[1]+1}]
print(confusion_matrix(parts["test"].target, pred))

3. La vectorización neuronal también aprende solo de train

MAX_TOKENS, MAX_LEN, DIM = 2500, 80, 24
vectorizer = keras.layers.TextVectorization(max_tokens=MAX_TOKENS,
    output_mode="int", output_sequence_length=MAX_LEN)
vectorizer.adapt(parts["train"].document.to_numpy())
vocab = vectorizer.get_vocabulary(); V = len(vocab)
Xn = {name: vectorizer(frame.document.to_numpy()).numpy() for name, frame in parts.items()}
yn = {name: frame.target.to_numpy() for name, frame in parts.items()}
print("vocabulario:", V, "shape train:", Xn["train"].shape)

4. Coocurrencia + SVD produce una referencia distribucional

cooc = np.zeros((V, V), dtype=np.float32)
for sequence in Xn["train"]:
    ids = sequence[sequence > 1]
    for i, center in enumerate(ids):
        for context in ids[max(0,i-2):i+3]:
            if center != context:
                cooc[center, context] += 1
U, S, _ = np.linalg.svd(np.log1p(cooc), full_matrices=False)
pretrained = (U[:, :DIM] * np.sqrt(S[:DIM])).astype("float32")
print("matriz preentrenada:", pretrained.shape)

5. La misma cabeza permite aislar la representación

def mean_model(initial=None, trainable=True, seed=SEED):
    keras.utils.set_random_seed(seed)
    emb = keras.layers.Embedding(V, DIM, mask_zero=True, trainable=trainable, name="embedding")
    model = keras.Sequential([keras.Input((MAX_LEN,), dtype="int32"), emb,
                              keras.layers.GlobalAveragePooling1D(),
                              keras.layers.Dense(1, activation="sigmoid")])
    emb.build((None, MAX_LEN))
    if initial is not None: emb.set_weights([initial])
    model.compile(optimizer=keras.optimizers.Adam(.01), loss="binary_crossentropy")
    return model

6. Tres regímenes, un mismo presupuesto

predictions = {}
for name, initial, trainable in [
    ("Aleatorio ajustable", None, True),
    ("SVD congelado", pretrained, False),
    ("SVD ajustable", pretrained, True),
]:
    model = mean_model(initial, trainable)
    t0 = time.perf_counter()
    model.fit(Xn["train"], yn["train"], epochs=3, batch_size=64, verbose=0,
              validation_data=(Xn["validation"], yn["validation"]))
    seconds = time.perf_counter()-t0
    pred = (model.predict(Xn["test"], verbose=0).ravel() >= .5).astype(int)
    predictions[name] = pred
    results.append({"modelo": name, "macro_f1": f1_score(yn["test"], pred, average="macro"),
                    "segundos": seconds,
                    "parámetros": int(sum(np.prod(w.shape) for w in model.trainable_weights))})
display(pd.DataFrame(results).sort_values("macro_f1", ascending=False).round(3))

7. El promedio y la LSTM no explotan el orden igual

keras.utils.set_random_seed(SEED)
lstm = keras.Sequential([keras.Input((MAX_LEN,), dtype="int32"),
    keras.layers.Embedding(V, DIM, mask_zero=True), keras.layers.LSTM(12),
    keras.layers.Dense(1, activation="sigmoid")])
lstm.compile(optimizer=keras.optimizers.Adam(.005), loss="binary_crossentropy")
t0=time.perf_counter(); lstm.fit(Xn["train"], yn["train"], epochs=2, batch_size=64,
    verbose=0, validation_data=(Xn["validation"], yn["validation"])); elapsed=time.perf_counter()-t0
lstm_pred=(lstm.predict(Xn["test"], verbose=0).ravel()>=.5).astype(int)
results.append({"modelo":"Embedding + LSTM", "macro_f1":f1_score(yn["test"],lstm_pred,average="macro"),
                "segundos":elapsed,"parámetros":lstm.count_params()})
display(pd.DataFrame(results).sort_values("macro_f1",ascending=False).round(3))

8. La auditoría busca atajos y subgrupos

audit = parts["test"][["topic", "source", "document", "target"]].copy()
audit["pred"] = lstm_pred
audit["longitud"] = audit.document.str.split().str.len()
audit["grupo_longitud"] = pd.qcut(audit.longitud, 3, labels=["corto","medio","largo"])
by_length = audit.groupby("grupo_longitud", observed=True).apply(
    lambda g: f1_score(g.target, g.pred, average="macro"), include_groups=False)
display(by_length.rename("macro_f1"))

9. Tres semillas separan resultado y estabilidad

seed_scores=[]
for seed in [17,42,73]:
    model=mean_model(pretrained, True, seed)
    model.fit(Xn["train"],yn["train"],epochs=2,batch_size=64,verbose=0)
    pred=(model.predict(Xn["test"],verbose=0).ravel()>=.5).astype(int)
    seed_scores.append(f1_score(yn["test"],pred,average="macro"))
print("macro-F1 por semilla:",np.round(seed_scores,3),"media/desv:",np.mean(seed_scores),np.std(seed_scores))

Dictamen

Redacte cinco párrafos: protocolo; construcción de cada representación; resultado frente al baseline; error por subgrupo; recomendación condicionada. Un embedding no queda validado únicamente porque sus vecinos “parecen razonables”.
