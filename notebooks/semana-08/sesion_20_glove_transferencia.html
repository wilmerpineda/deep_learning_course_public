Sesión 20 — GloVe, transferencia y LSTM

Curso: Deep Learning · Universidad Externado de Colombia

Propósito y evidencia

Construiremos coocurrencias globales, entrenaremos un GloVe pequeño y transferiremos sus vectores a una LSTM en tres regímenes: congelado, ajustable y aleatorio.

import os
os.environ["TF_CPP_MIN_LOG_LEVEL"] = "2"
from pathlib import Path
from collections import Counter
import re
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow import keras

SEED = 42
keras.utils.set_random_seed(SEED)
rng = np.random.default_rng(SEED)

1. El corpus real conserva sus particiones

def locate_data():
    candidates = [Path("datasets/public/spanish_fake_news.csv"),
                  Path("../../datasets/public/spanish_fake_news.csv")]
    return next(path for path in candidates if path.exists())

data = pd.read_csv(locate_data()).fillna("")
data["document"] = (data["headline"] + " " + data["text"]).str.strip()
data["target"] = (data["label"] == "true").astype("float32")
display(data.groupby(["split", "label"]).size().unstack(fill_value=0))

2. El vocabulario solo aprende de train

def tokenize(text):
    return re.findall(r"[a-záéíóúüñ]+", text.lower())

train_tokens = [tokenize(text) for text in data.loc[data.split == "train", "document"]]
counts = Counter(token for row in train_tokens for token in row)
vocabulary = ["<pad>", "<unk>"] + [word for word, _ in counts.most_common(498)]
token_to_id = {token: i for i, token in enumerate(vocabulary)}
print("vocabulario:", len(vocabulary), "tokens train:", sum(map(len, train_tokens)))

3. Una matriz resume coocurrencias

V, window = len(vocabulary), 2
cooc = np.zeros((V, V), dtype="float32")
for tokens in train_tokens:
    ids = [token_to_id.get(token, 1) for token in tokens[:200]]
    for i, center in enumerate(ids):
        for j in range(max(0, i-window), min(len(ids), i+window+1)):
            if i != j:
                cooc[center, ids[j]] += 1 / abs(i-j)
rows, cols = np.nonzero(cooc)
values = cooc[rows, cols]
print("shape:", cooc.shape, "pares no nulos:", len(values))

4. GloVe aproxima el logaritmo de la coocurrencia

Cada término combina dos vectores y dos sesgos. La función de peso reduce la influencia de pares extremadamente frecuentes.

dim = 16
W = tf.Variable(rng.normal(0, .05, (V, dim)).astype("float32"))
C = tf.Variable(rng.normal(0, .05, (V, dim)).astype("float32"))
bw = tf.Variable(tf.zeros(V)); bc = tf.Variable(tf.zeros(V))
ri = tf.constant(rows, dtype=tf.int32); ci = tf.constant(cols, dtype=tf.int32)
x = tf.constant(values, dtype=tf.float32)
weights = tf.minimum((x / 50.0) ** 0.75, 1.0)
optimizer = keras.optimizers.Adam(0.03)

losses = []
for _ in range(35):
    with tf.GradientTape() as tape:
        prediction = tf.reduce_sum(tf.gather(W, ri) * tf.gather(C, ci), axis=1)
        prediction += tf.gather(bw, ri) + tf.gather(bc, ci)
        loss = tf.reduce_mean(weights * tf.square(prediction - tf.math.log(x)))
    variables = [W, C, bw, bc]
    optimizer.apply_gradients(zip(tape.gradient(loss, variables), variables))
    losses.append(float(loss))
print("pérdida inicial/final:", losses[0], losses[-1])
assert losses[-1] < losses[0]
glove_vectors = (W + C).numpy()

5. Las secuencias respetan padding y desconocidos

MAX_LEN = 60
def encode(text):
    ids = [token_to_id.get(token, 1) for token in tokenize(text)[:MAX_LEN]]
    return ids + [0] * (MAX_LEN-len(ids))

X = np.asarray([encode(text) for text in data.document], dtype="int32")
y = data.target.to_numpy()
masks = {split: data.split.eq(split).to_numpy() for split in ["train", "validation", "test"]}
print(X.shape, {key: int(value.sum()) for key, value in masks.items()})

6. Transferir es copiar una matriz con un contrato de índices

def text_lstm(initial_weights=None, trainable=True, seed=SEED):
    keras.utils.set_random_seed(seed)
    layer = keras.layers.Embedding(V, dim, mask_zero=True, name="embedding",
                                   trainable=trainable)
    model = keras.Sequential([
        keras.Input((MAX_LEN,), dtype="int32"), layer,
        keras.layers.LSTM(8), keras.layers.Dense(1, activation="sigmoid")])
    layer.build((None, MAX_LEN))
    if initial_weights is not None:
        layer.set_weights([initial_weights])
    model.compile(optimizer=keras.optimizers.Adam(0.005),
                  loss="binary_crossentropy", metrics=["accuracy"])
    return model

frozen = text_lstm(glove_vectors, trainable=False)
before = frozen.get_layer("embedding").get_weights()[0].copy()
frozen.fit(X[masks["train"]], y[masks["train"]], epochs=1, batch_size=64, verbose=0)
after = frozen.get_layer("embedding").get_weights()[0]
np.testing.assert_allclose(before, after)
print("embedding congelado: cambio máximo", np.abs(before-after).max())

7. Congelado, fine-tuning y aleatorio responden preguntas distintas

rows_out = []
for name, initial, trainable in [
    ("GloVe congelado", glove_vectors, False),
    ("GloVe ajustable", glove_vectors, True),
    ("Aleatorio ajustable", None, True),
]:
    model = text_lstm(initial, trainable)
    model.fit(X[masks["train"]], y[masks["train"]], epochs=2,
              batch_size=64, verbose=0,
              validation_data=(X[masks["validation"]], y[masks["validation"]]))
    loss, accuracy = model.evaluate(X[masks["test"]], y[masks["test"]], verbose=0)
    rows_out.append({"régimen": name, "accuracy_test": accuracy,
                     "parámetros_entrenables": int(sum(np.prod(v.shape) for v in model.trainable_weights))})
display(pd.DataFrame(rows_out).round(3))

8. La comparación es evidencia local, no un ranking universal

coverage = np.mean([token in token_to_id for text in data.loc[data.split == "test", "document"]
                    for token in tokenize(text)])
print("cobertura de tokens de test:", round(float(coverage), 3))
assert 0 <= coverage <= 1
