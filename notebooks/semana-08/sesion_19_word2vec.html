Sesión 19 — Word2Vec y contexto

Curso: Deep Learning · Universidad Externado de Colombia

Propósito y evidencia

Convertiremos texto en pares centro–contexto, calcularemos una actualización Skip-gram y entrenaremos dos tablas de vectores. La evidencia será la pérdida, los scores de pares observados y las vecindades resultantes.

import os
os.environ["TF_CPP_MIN_LOG_LEVEL"] = "2"
from collections import Counter
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow import keras

SEED = 42
keras.utils.set_random_seed(SEED)
rng = np.random.default_rng(SEED)

1. Un corpus visible

La ventana de contexto define qué observaciones alimentan el objetivo. No existe una noción de cercanía independiente de esa decisión.

sentences = [
    "café colombiano aumenta exportaciones", "precio del café sube",
    "ventas de café crecen", "demanda internacional aumenta",
    "café colombiano reduce exportaciones", "precio del café cae",
    "ventas de café disminuyen", "demanda internacional baja",
    "economía colombiana crece", "economía colombiana se desacelera",
]
tokenized = [sentence.split() for sentence in sentences]
counts = Counter(token for row in tokenized for token in row)
vocabulary = ["<unk>"] + sorted(counts)
token_to_id = {token: i for i, token in enumerate(vocabulary)}
encoded = [[token_to_id[token] for token in row] for row in tokenized]
print("vocabulario:", len(vocabulary), vocabulary)

2. La ventana produce pares centro–contexto

def positive_pairs(sequences, window=2):
    pairs = []
    for sequence in sequences:
        for i, center in enumerate(sequence):
            left, right = max(0, i-window), min(len(sequence), i+window+1)
            pairs.extend((center, sequence[j]) for j in range(left, right) if j != i)
    return pairs

positives = positive_pairs(encoded, window=2)
preview = [(vocabulary[c], vocabulary[o]) for c, o in positives[:10]]
display(pd.DataFrame(preview, columns=["centro", "contexto"]))

3. Negative sampling crea una tarea binaria

Un par observado recibe etiqueta 1. Para cada positivo muestreamos contextos no observados y les asignamos 0. La tarea auxiliar organiza la geometría.

def training_examples(positives, vocab_size, negatives_per_positive=2):
    examples = []
    observed = set(positives)
    for center, context in positives:
        examples.append((center, context, 1.0))
        candidates = [i for i in range(1, vocab_size)
                      if i != context and (center, i) not in observed]
        for negative in rng.choice(candidates, size=negatives_per_positive, replace=True):
            examples.append((center, int(negative), 0.0))
    rng.shuffle(examples)
    return np.asarray(examples, dtype=np.float32)

examples = training_examples(positives, len(vocabulary))
centers = examples[:, 0].astype("int32")
contexts = examples[:, 1].astype("int32")
targets = examples[:, 2]
print("positivos/negativos:", np.bincount(targets.astype(int)))

4. Un ejemplo manual abre la caja

El score es el producto punto entre una fila de la tabla central y una fila de la tabla contextual. BCE empuja hacia arriba un par positivo y hacia abajo uno negativo.

u = np.array([0.2, -0.1, 0.4])
v = np.array([0.5, 0.3, -0.2])
score = float(u @ v)
probability = 1 / (1 + np.exp(-score))
loss_positive = -np.log(probability)
gradient_u = (probability - 1.0) * v
print({"score": score, "p(y=1)": probability, "BCE": loss_positive})
print("gradiente respecto de u:", gradient_u)

5. Skip-gram aprende dos tablas

dim = 8
center_input = keras.Input(shape=(), dtype="int32", name="centro")
context_input = keras.Input(shape=(), dtype="int32", name="contexto")
center_table = keras.layers.Embedding(len(vocabulary), dim, name="tabla_centro")
context_table = keras.layers.Embedding(len(vocabulary), dim, name="tabla_contexto")
center_vector = center_table(center_input)
context_vector = context_table(context_input)
logit = keras.layers.Dot(axes=-1)([center_vector, context_vector])
skipgram = keras.Model([center_input, context_input], logit)
skipgram.compile(
    optimizer=keras.optimizers.Adam(0.03),
    loss=keras.losses.BinaryCrossentropy(from_logits=True),
)
history = skipgram.fit([centers, contexts], targets, epochs=60, batch_size=32, verbose=0)
print("pérdida inicial/final:", history.history["loss"][0], history.history["loss"][-1])
assert history.history["loss"][-1] < history.history["loss"][0]

6. Los pares observados deben recibir scores mayores

scores = skipgram.predict([centers, contexts], verbose=0).ravel()
summary = pd.DataFrame({"score": scores, "objetivo": targets}).groupby("objetivo").mean()
display(summary)
assert summary.loc[1.0, "score"] > summary.loc[0.0, "score"]

7. Una sola tabla se usa como representación

vectors = center_table.get_weights()[0]
def nearest(token, k=4):
    query = vectors[token_to_id[token]]
    norms = np.linalg.norm(vectors, axis=1) * np.linalg.norm(query)
    similarity = vectors @ query / np.maximum(norms, 1e-8)
    order = np.argsort(-similarity)
    return [(vocabulary[i], float(similarity[i])) for i in order if vocabulary[i] != token][:k]

for token in ["café", "aumenta", "disminuyen", "economía"]:
    print(token, nearest(token))

8. CBOW invierte la pregunta

CBOW promedia los vectores del contexto para predecir el centro. Skip-gram usa el centro para discriminar contextos. Comparten la señal distribucional, no la arquitectura exacta.

context_ids = np.array([[token_to_id["precio"], token_to_id["del"], token_to_id["sube"]]])
context_vectors = vectors[context_ids]
cbow_representation = context_vectors.mean(axis=1)
assert cbow_representation.shape == (1, dim)
print("representación CBOW:", cbow_representation.round(3))
